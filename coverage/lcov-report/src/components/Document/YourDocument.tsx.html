        confirmText={isUploaded ? "Elimina" : "Rimuovi"}
        cancelText="Annulla"
        onCancel={cancelRemove}
        onConfirm={confirmRemove}
      />
 
      <DocumentRenameModal
        isOpen={isRenameOpen}
        itemToRename={itemToRename}
        newName={newName}
        setNewName={setNewName}
        onClose={closeRenameModal}
        onSubmit={handleRenameSubmit}
      />
    </section>
  );
};
